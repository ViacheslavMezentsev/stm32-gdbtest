cmake_minimum_required(VERSION 3.25)
project(gdbtest_cpp_context LANGUAGES C)
get_filename_component(compiler_dir "${CMAKE_C_COMPILER}" DIRECTORY)
if(CMAKE_HOST_WIN32)
    set(CMAKE_CXX_COMPILER "${compiler_dir}/arm-none-eabi-g++.exe" CACHE FILEPATH "ARM C++ compiler")
else()
    set(CMAKE_CXX_COMPILER "${compiler_dir}/arm-none-eabi-g++" CACHE FILEPATH "ARM C++ compiler")
endif()
enable_language(CXX)
include(CTest)
get_filename_component(module "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CI_PROFILE f030r8 CACHE STRING "MCU profile")
set(profiles f030r8 f103c8 f401cc f411ce f429zi at32f403a)
set_property(CACHE CI_PROFILE PROPERTY STRINGS ${profiles})
if(NOT CI_PROFILE IN_LIST profiles)
    message(FATAL_ERROR "Unknown CI_PROFILE: ${CI_PROFILE}")
endif()
set(OPT Og CACHE STRING "Optimization: Og or O2")
set_property(CACHE OPT PROPERTY STRINGS Og O2)
if(NOT OPT MATCHES "^(Og|O2)$")
    message(FATAL_ERROR "OPT must be Og or O2")
endif()
if(CI_PROFILE STREQUAL "f030r8")
    set(cpu cortex-m0)
elseif(CI_PROFILE STREQUAL "f103c8")
    set(cpu cortex-m3)
else()
    set(cpu cortex-m4)
endif()
set(profile "${module}/tests/firmware/profiles/${CI_PROFILE}")
add_executable(cpp_context src/startup.c src/cpp_main.cpp src/cpp_values.cpp)
set_target_properties(cpp_context PROPERTIES SUFFIX .elf)
set(arch -mcpu=${cpu} -mthumb -mfloat-abi=soft)
target_compile_options(cpp_context PRIVATE ${arch} -${OPT} -g3 -fno-lto -ffunction-sections -fdata-sections
    $<$<COMPILE_LANGUAGE:C>:-fno-tree-loop-distribute-patterns>
    $<$<COMPILE_LANGUAGE:CXX>:-std=c++17;-fno-exceptions;-fno-rtti> -Wall -Wextra -Werror)
target_link_options(cpp_context PRIVATE ${arch} -nostdlib "-T${profile}/firmware_FLASH.ld" -Wl,--gc-sections)
target_link_libraries(cpp_context PRIVATE gcc)
include("${module}/stm32_gdbtest/cmake/STM32GDBTest.cmake")
stm32_gdbtest_attach(cpp_context PROFILE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/profile"
    SESSION_CONFIG "${profile}/session.toml"
    MANIFEST_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${profile}/firmware_FLASH.ld"
    "${module}/tests/firmware/cmake/arm-gcc.cmake")
