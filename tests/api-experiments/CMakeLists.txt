cmake_minimum_required(VERSION 3.25)
project(rc3_api_experiments LANGUAGES C)
include(CTest)
find_package(Python3 3.11 COMPONENTS Interpreter REQUIRED)
get_filename_component(MODULE "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(RC3_OPTIMIZED_FIXTURE "Build the ordinary dual-channel calculation application" OFF)
option(RC3_ABI_FIXTURE "Build the ordinary mixed-type calculation application" OFF)
if(RC3_ABI_FIXTURE AND RC3_OPTIMIZED_FIXTURE)
    message(FATAL_ERROR "Select only one application")
endif()
if(RC3_ABI_FIXTURE)
    add_executable(api_firmware src/startup.c src/abi_main.c src/abi_values.c)
elseif(RC3_OPTIMIZED_FIXTURE)
    add_executable(api_firmware src/startup.c src/optimized.c)
else()
    add_executable(api_firmware src/startup.c src/app.c src/processing.c src/packet.c src/returns.c src/routes.c)
endif()
set_target_properties(api_firmware PROPERTIES SUFFIX .elf)
set(RC3_FLOAT_ABI "soft" CACHE STRING "Research floating-point ABI: soft or hard")
set_property(CACHE RC3_FLOAT_ABI PROPERTY STRINGS soft hard)
if(NOT RC3_FLOAT_ABI MATCHES "^(soft|hard)$")
    message(FATAL_ERROR "RC3_FLOAT_ABI must be soft or hard")
endif()
set(arch -mcpu=cortex-m4 -mthumb -mfloat-abi=${RC3_FLOAT_ABI})
if(RC3_FLOAT_ABI STREQUAL "hard")
    if(NOT RC3_ABI_FIXTURE)
        message(FATAL_ERROR "Hard-float requires the application with FPU initialization")
    endif()
    list(APPEND arch -mfpu=fpv4-sp-d16)
endif()
set(RC3_OPTIMIZATION "Og" CACHE STRING "Research firmware optimization: Og, O2 or Os")
set_property(CACHE RC3_OPTIMIZATION PROPERTY STRINGS Og O2 Os)
if(NOT RC3_OPTIMIZATION MATCHES "^(Og|O2|Os)$")
    message(FATAL_ERROR "RC3_OPTIMIZATION must be Og, O2 or Os")
endif()
if(NOT RC3_OPTIMIZATION STREQUAL "Og")
    # Startup runs before the C runtime; keep its copy/zero loops self-contained.
    set_source_files_properties(src/startup.c PROPERTIES COMPILE_OPTIONS -fno-tree-loop-distribute-patterns)
endif()
target_compile_options(api_firmware PRIVATE ${arch} -std=c11 -${RC3_OPTIMIZATION} -g3 -fno-lto
    -ffunction-sections -fdata-sections -Wall -Wextra -Werror)
target_link_options(api_firmware PRIVATE ${arch} -fno-lto -nostdlib
    "-T${CMAKE_CURRENT_SOURCE_DIR}/firmware_FLASH.ld" -Wl,--gc-sections)
target_link_libraries(api_firmware PRIVATE gcc)
include("${MODULE}/stm32_gdbtest/cmake/STM32GDBTest.cmake")
stm32_gdbtest_attach(api_firmware PROFILE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/profile"
    MANIFEST_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/firmware_FLASH.ld"
    "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" "${MODULE}/tests/firmware/cmake/arm-gcc.cmake")
add_test(NAME host.research COMMAND "${Python3_EXECUTABLE}" -B -m unittest discover
    -s "${CMAKE_CURRENT_SOURCE_DIR}/host" -v)
set_tests_properties(host.research PROPERTIES LABELS host
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
